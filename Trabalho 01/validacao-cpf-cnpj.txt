//Função para validar o CPF
    function valida_cpf($cpf)
    {
        // Remove caracteres não numéricos (boa prática)
        $cpf = preg_replace('/[^0-9]/', '', $cpf);

        $digitos_iguais = 1;

        if (strlen($cpf) < 11)
            return false;

        // Verifica se todos os dígitos são iguais
        for ($i = 0; $i < strlen($cpf) - 1; $i++) {
            if ($cpf[$i] != $cpf[$i + 1]) {
                $digitos_iguais = 0;
                break;
            }
        }

        if (!$digitos_iguais) {

            $numeros = substr($cpf, 0, 9);
            $digitos = substr($cpf, 9, 2);

            $soma = 0;
            for ($i = 10; $i > 1; $i--) {
                $soma += $numeros[10 - $i] * $i;
            }

            $resultado = ($soma % 11 < 2) ? 0 : 11 - ($soma % 11);

            if ($resultado != $digitos[0])
                return false;

            $numeros = substr($cpf, 0, 10);
            $soma = 0;

            for ($i = 11; $i > 1; $i--) {
                $soma += $numeros[11 - $i] * $i;
            }

            $resultado = ($soma % 11 < 2) ? 0 : 11 - ($soma % 11);

            if ($resultado != $digitos[1])
                return false;

            return true;
        } else {
            return false;
        }
    }

    //Função para validar o CNPJ
    function valida_cnpj($cnpj)
    {
        // Remove caracteres não numéricos (boa prática)
        $cnpj = preg_replace('/[^0-9]/', '', $cnpj);

        $digitos_iguais = 1;

        if (strlen($cnpj) < 14)
            return false;

        // Verifica se todos os dígitos são iguais
        for ($i = 0; $i < strlen($cnpj) - 1; $i++) {
            if ($cnpj[$i] != $cnpj[$i + 1]) {
                $digitos_iguais = 0;
                break;
            }
        }

        if (!$digitos_iguais) {

            $tamanho = strlen($cnpj) - 2;
            $numeros = substr($cnpj, 0, $tamanho);
            $digitos = substr($cnpj, $tamanho, 2);

            $soma = 0;
            $pos = $tamanho - 7;

            for ($i = $tamanho; $i >= 1; $i--) {
                $soma += $numeros[$tamanho - $i] * $pos--;
                if ($pos < 2)
                    $pos = 9;
            }

            $resultado = ($soma % 11 < 2) ? 0 : 11 - ($soma % 11);

            if ($resultado != $digitos[0])
                return false;

            $tamanho = $tamanho + 1;
            $numeros = substr($cnpj, 0, $tamanho);

            $soma = 0;
            $pos = $tamanho - 7;

            for ($i = $tamanho; $i >= 1; $i--) {
                $soma += $numeros[$tamanho - $i] * $pos--;
                if ($pos < 2)
                    $pos = 9;
            }

            $resultado = ($soma % 11 < 2) ? 0 : 11 - ($soma % 11);

            if ($resultado != $digitos[1])
                return false;

            return true;
        } else {
            return false;
        }
    }
